import { Injectable } from '@nestjs/common';
import { SynthClient } from './synth.client';
@Injectable()
export class TextToSpeechResponseService {
constructor(private readonly synth: SynthClient) {}
async generateAudio(text: string): Promise<Buffer> {
return await this.withRetry(() => this.synth.generate(text));
}
private async withRetry<T>(
fn: () => Promise<T>,
attempts = 3,
): Promise<T> {
for (let i = 0; i < attempts; i++) {
try { return await fn(); }
catch (e: any) {
if (e.status === 429 && i < attempts - 1) {
await this.sleep(500 * 2 ** i);
continue;
}
throw e;
}
}
throw new Error('unreachable');
}
private sleep(ms: number): Promise<void> {
return new Promise(r => setTimeout(r, ms));
}
}
